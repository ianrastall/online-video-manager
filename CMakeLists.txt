cmake_minimum_required(VERSION 3.25)
project(OnlineVideoManager VERSION 0.2.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "OVM's native platform implementation requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(miniz URL https://github.com/richgel999/miniz/archive/refs/tags/3.1.0.zip
  URL_HASH SHA256=89ce8fc6012b1f48407bb1422dd75701bb639e00fc235090cd8b47283fe6197b)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json miniz)
add_library(ovm_engine STATIC native/platform.cpp native/media.cpp native/tools.cpp native/engine.cpp)
target_include_directories(ovm_engine PUBLIC native include)
target_link_libraries(ovm_engine PUBLIC nlohmann_json::nlohmann_json miniz winhttp bcrypt shell32 ole32)
target_compile_definitions(ovm_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
if(MSVC)
  target_compile_options(ovm_engine PRIVATE /W4 /utf-8 /permissive- /EHsc)
endif()
add_library(ovm_core SHARED native/abi.cpp)
target_link_libraries(ovm_core PRIVATE ovm_engine)
target_compile_definitions(ovm_core PRIVATE OVM_BUILD)
set_target_properties(ovm_core PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/native")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(ovm_core PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/artifacts/native")
endforeach()
enable_testing()
add_executable(ovm_test_process native/test_process.cpp)
add_executable(ovm_native_tests native/tests.cpp)
target_link_libraries(ovm_native_tests PRIVATE ovm_engine ovm_core)
add_dependencies(ovm_native_tests ovm_test_process)
if(MSVC)
  target_compile_options(ovm_native_tests PRIVATE /utf-8)
endif()
add_test(NAME native_engine COMMAND ovm_native_tests $<TARGET_FILE:ovm_test_process>)
set_tests_properties(native_engine PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${CMAKE_SOURCE_DIR}/artifacts/native")
set_tests_properties(native_engine PROPERTIES TIMEOUT 45)
add_executable(ovm_c_abi_smoke native/c_abi_smoke.c)
target_include_directories(ovm_c_abi_smoke PRIVATE include)
target_link_libraries(ovm_c_abi_smoke PRIVATE ovm_core)
add_test(NAME c_abi_client COMMAND ovm_c_abi_smoke "${CMAKE_BINARY_DIR}/c-client-home")
set_tests_properties(c_abi_client PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${CMAKE_SOURCE_DIR}/artifacts/native")
install(TARGETS ovm_core RUNTIME DESTINATION .)
